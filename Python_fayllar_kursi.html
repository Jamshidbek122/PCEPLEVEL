    hisobot.py
    import csv
import json
from pathlib import Path
from datetime import date

KIRISH = Path("baholar.csv")
CHIQISH = Path("natijalar")


def baho_harfi(ortacha):
    if ortacha >= 86: return "A"
    if ortacha >= 71: return "B"
    if ortacha >= 56: return "C"
    return "F"


def oqish(yol):
    try:
        with open(yol, newline="", encoding="utf-8") as f:
            return list(csv.DictReader(f))
    except FileNotFoundError:
        print(f"'{yol}' topilmadi!")
        return []


def hisoblash(qatorlar):
    natijalar = []
    for q in qatorlar:
        ism = q.pop("ism")
        ballar = [int(b) for b in q.values()]
        ortacha = round(sum(ballar) / len(ballar), 1)
        natijalar.append({"ism": ism, "ortacha": ortacha, "baho": baho_harfi(ortacha)})
    natijalar.sort(key=lambda n: n["ortacha"], reverse=True)
    return natijalar


def saqlash(natijalar):
    CHIQISH.mkdir(exist_ok=True)

    with open(CHIQISH / "hisobot.txt", "w", encoding="utf-8") as f:
        f.write(f"BAHOLAR HISOBOTI: {date.today()}\n")
        f.write("=" * 35 + "\n")
        for i, n in enumerate(natijalar, 1):
            f.write(f"{i}. {n['ism']:<10} {n['ortacha']:>6}   {n['baho']}\n")

    with open(CHIQISH / "natijalar.json", "w", encoding="utf-8") as f:
        json.dump(natijalar, f, ensure_ascii=False, indent=2)

    with open(CHIQISH / "natijalar.csv", "w", newline="", encoding="utf-8-sig") as f:
        writer = csv.DictWriter(f, fieldnames=["ism", "ortacha", "baho"])
        writer.writeheader()
        writer.writerows(natijalar)


if __name__ == "__main__":
    qatorlar = oqish(KIRISH)
    if qatorlar:
